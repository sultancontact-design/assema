// src/components/social/follow-button.tsx
'use client'
import { useState } from 'react'
import { Button } from '@/components/ui/button'
import { UserPlus, UserCheck } from 'lucide-react'
import { toast } from 'sonner'

export function FollowButton({ 
  userId, 
  initialFollowing = false,
  size = 'default' 
}: { 
  userId: string
  initialFollowing?: boolean
  size?: 'default' | 'sm'
}) {
  const [following, setFollowing] = useState(initialFollowing)
  const [loading, setLoading] = useState(false)
  
  const handleToggle = async () => {
    setLoading(true)
    try {
      const res = await fetch(`/api/users/${userId}/follow`, {
        method: following ? 'DELETE' : 'POST',
      })
      if (res.ok) {
        setFollowing(!following)
        toast.success(following ? 'تم إلغاء المتابعة' : 'تمت المتابعة')
      }
    } finally {
      setLoading(false)
    }
  }
  
  return (
    <Button
      onClick={handleToggle}
      disabled={loading}
      variant={following ? 'outline' : 'default'}
      size={size}
      className="gap-2"
    >
      {following ? (
        <>
          <UserCheck className="h-4 w-4" />
          متابَع
        </>
      ) : (
        <>
          <UserPlus className="h-4 w-4" />
          متابعة
        </>
      )}
    </Button>
  )
}

【API Endpoint】
```typescript
// src/app/api/users/[id]/follow/route.ts
import { prisma } from '@/lib/db'
import { getCurrentUserId } from '@/lib/auth'

export async function POST(req: Request, { params }: { params: { id: string } }) {
  const currentUserId = await getCurrentUserId()
  if (!currentUserId) return Response.json({ error: 'Unauthorized' }, { status: 401 })
  
  await prisma.follow.create({
    data: {
      followerId: currentUserId,
      followingId: params.id,
    },
  })
  
  // إشعار للمستخدم
  await prisma.notification.create({
    data: {
      userId: params.id,
      type: 'FOLLOW',
      title: 'متابع جديد',
      body: 'بدأ أحدهم بمتابعتك',
    },
  })
  
  return Response.json({ success: true })
}

export async function DELETE(req: Request, { params }: { params: { id: string } }) {
  const currentUserId = await getCurrentUserId()
  if (!currentUserId) return Response.json({ error: 'Unauthorized' }, { status: 401 })
  
  await prisma.follow.deleteMany({
    where: {
      followerId: currentUserId,
      followingId: params.id,
    },
  })
  
  return Response.json({ success: true })
}

// src/components/social/comment-section.tsx
'use client'
import { useState, useEffect } from 'react'
import { Avatar, AvatarImage, AvatarFallback } from '@/components/ui/avatar'
import { Button } from '@/components/ui/button'
import { Textarea } from '@/components/ui/textarea'
import { Heart, Reply, MoreHorizontal } from 'lucide-react'
import { toast } from 'sonner'

export function CommentSection({ postId }: { postId: string }) {
  const [comments, setComments] = useState([])
  const [newComment, setNewComment] = useState('')
  const [loading, setLoading] = useState(false)
  
  useEffect(() => {
    fetch(`/api/posts/${postId}/comments`)
      .then(r => r.json())
      .then(d => setComments(d.comments || []))
  }, [postId])
  
  const handleSubmit = async () => {
    if (!newComment.trim()) return
    setLoading(true)
    try {
      const res = await fetch(`/api/posts/${postId}/comments`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ content: newComment }),
      })
      const data = await res.json()
      setComments([data.comment, ...comments])
      setNewComment('')
      toast.success('تم إضافة التعليق')
    } finally {
      setLoading(false)
    }
  }
  
  return (
    <div className="space-y-4" dir="rtl">
      {/* نموذج إضافة تعليق */}
      <div className="flex gap-3">
        <Avatar className="h-10 w-10">
          <AvatarFallback>أ</AvatarFallback>
        </Avatar>
        <div className="flex-1 space-y-2">
          <Textarea
            placeholder="أضف تعليقاً..."
            value={newComment}
            onChange={(e) => setNewComment(e.target.value)}
            rows={2}
            className="resize-none"
          />
          <div className="flex justify-end">
            <Button 
              onClick={handleSubmit} 
              disabled={loading || !newComment.trim()}
              size="sm"
            >
              {loading ? 'جاري الإرسال...' : 'تعليق'}
            </Button>
          </div>
        </div>
      </div>
      
      {/* قائمة التعليقات */}
      <div className="space-y-4">
        {comments.map((comment: any) => (
          <CommentItem key={comment.id} comment={comment} />
        ))}
      </div>
    </div>
  )
}

function CommentItem({ comment }: { comment: any }) {
  const [liked, setLiked] = useState(comment.isLiked)
  const [likes, setLikes] = useState(comment.likes)
  
  return (
    <div className="flex gap-3">
      <Avatar className="h-8 w-8">
        <AvatarImage src={comment.author.avatar} />
        <AvatarFallback>{comment.author.name[0]}</AvatarFallback>
      </Avatar>
      <div className="flex-1">
        <div className="bg-muted rounded-lg p-3">
          <div className="font-semibold text-sm">{comment.author.name}</div>
          <p className="text-sm mt-1">{comment.content}</p>
        </div>
        <div className="flex gap-4 mt-2 text-sm text-muted-foreground">
          <button 
            className="hover:text-primary flex items-center gap-1"
            onClick={() => {
              setLiked(!liked)
              setLikes(liked ? likes - 1 : likes + 1)
            }}
          >
            <Heart className={`h-3 w-3 ${liked ? 'fill-red-500 text-red-500' : ''}`} />
            {likes}
          </button>
          <button className="hover:text-primary flex items-center gap-1">
            <Reply className="h-3 w-3" />
            رد
          </button>
        </div>
      </div>
    </div>
  )
}

// src/components/social/share-menu.tsx
'use client'
import { Share2, Facebook, Twitter, Linkedin, Copy, MessageCircle } from 'lucide-react'
import { Button } from '@/components/ui/button'
import { DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuTrigger } from '@/components/ui/dropdown-menu'
import { toast } from 'sonner'

export function ShareMenu({ url, title }: { url: string; title: string }) {
  const shareUrl = typeof window !== 'undefined' ? window.location.origin + url : url
  
  const handleShare = (platform: string) => {
    const encodedUrl = encodeURIComponent(shareUrl)
    const encodedTitle = encodeURIComponent(title)
    
    const urls: Record<string, string> = {
      facebook: `https://www.facebook.com/sharer/sharer.php?u=${encodedUrl}`,
      twitter: `https://twitter.com/intent/tweet?url=${encodedUrl}&text=${encodedTitle}`,
      linkedin: `https://www.linkedin.com/sharing/share-offsite/?url=${encodedUrl}`,
      whatsapp: `https://wa.me/?text=${encodedTitle}%20${encodedUrl}`,
    }
    
    window.open(urls[platform], '_blank', 'width=600,height=400')
  }
  
  const handleCopy = () => {
    navigator.clipboard.writeText(shareUrl)
    toast.success('تم نسخ الرابط')
  }
  
  return (
    <DropdownMenu>
      <DropdownMenuTrigger asChild>
        <button className="flex items-center gap-1 hover:text-primary transition-colors">
          <Share2 className="h-4 w-4" />
          <span className="text-sm">مشاركة</span>

        <DropdownMenuItem onClick={() => handleShare('facebook')}>
          <Facebook className="h-4 w-4 ml-2" />
          فيسبوك
        </DropdownMenuItem>
        <DropdownMenuItem onClick={() => handleShare('twitter')}>
          <Twitter className="h-4 w-4 ml-2" />

```tsx
// src/components/social/video-player.tsx
'use client'
import { useState } from 'react'
import { Play } from 'lucide-react'

export function VideoPlayer({ 
  src, 
  poster, 
  type = 'youtube',
  className = ''
}: { 
  src: string
  poster?: string
  type?: 'youtube' | 'vimeo' | 'direct'
  className?: string
}) {
  const [playing, setPlaying] = useState(false)
  
  // استخراج ID من يوتيوب
  const getYouTubeId = (url: string) => {
    const match = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([^&]+)/)
    return match ? match[1] : url
  }
  
  const embedSrc = type === 'youtube' 
    ? `https://www.youtube.com/embed/${getYouTubeId(src)}?autoplay=1`
    : src
  
  return (
    <div className={`relative w-full aspect-video rounded-lg overflow-hidden bg-black ${className}`}>
      {!playing ? (
        <>
          {/* صورة الغلاف */}
          {poster ? (
            <img 
              src={poster} 
              alt="فيديو"
              className="w-full h-full object-cover"
            />
          ) : (
            <div className="w-full h-full bg-gradient-to-br from-primary/20 to-secondary/20" />
          )}
          
          {/* زر التشغيل */}
          <button
            onClick={() => setPlaying(true)}
            className="absolute inset-0 flex items-center justify-center group"
            aria-label="تشغيل الفيديو"
          >
            <div className="w-16 h-16 rounded-full bg-white/90 flex items-center justify-center group-hover:scale-110 transition-transform">
              <Play className="h-8 w-8 text-primary ml-1" fill="currentColor" />
            </div>
          </button>
        </>
      ) : (
        <iframe
          src={embedSrc}
          title="فيديو"
          allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share"
          allowFullScreen
          className="absolute inset-0 w-full h-full"
        />
      )}
    </div>
  )

// src/components/social/unified-feed.tsx
'use client'
import { useEffect, useState } from 'react'
import { PostCard } from './post-card'
import { ContributionCard } from './contribution-card'
import { EventCard } from './event-card'
import { BadgeCard } from './badge-card'
import { BlogCard } from './blog-card'
import { PollCard } from './poll-card'
import { VideoCard } from './video-card'

const CARD_COMPONENTS: Record<string, React.ComponentType<any>> = {
  POST: PostCard,
  CONTRIBUTION: ContributionCard,
  EVENT: EventCard,
  BADGE_EARNED: BadgeCard,
  LEVEL_UP: BadgeCard,
  BLOG_PUBLISHED: BlogCard,
  POLL: PollCard,
  VIDEO: VideoCard,
}

export function UnifiedFeed() {
  const [items, setItems] = useState([])
  const [loading, setLoading] = useState(true)
  
  useEffect(() => {
    fetch('/api/feed')
      .then(r => r.json())
      .then(d => setItems(d.items || []))
      .finally(() => setLoading(false))
  }, [])
  
  if (loading) return <FeedSkeleton />
  
  return (
    <div className="space-y-4">
      {items.map((item: any) => {
        const CardComponent = CARD_COMPONENTS[item.type] || PostCard
        return <CardComponent key={item.id} item={item} />
      })}
    </div>
  )
}

```tsx
// src/components/social/post-card.tsx
'use client'
import { Avatar, AvatarImage, AvatarFallback } from '@/components/ui/avatar'
import { Heart, MessageCircle, Share2, Bookmark, MoreHorizontal } from 'lucide-react'
import { FollowButton } from './follow-button'
import { CommentSection } from './comment-section'
import { BookmarkButton } from './bookmark-button'
import { ShareMenu } from './share-menu'
import { useState } from 'react'

export function PostCard({ item }: { item: any }) {
  const [liked, setLiked] = useState(item.isLiked)
  const [likes, setLikes] = useState(item.likes)
  const [showComments, setShowComments] = useState(false)
  
  return (
    <article className="bg-card border rounded-2xl p-4 space-y-3" dir="rtl">
      {/* Header */}
      <div className="flex items-center gap-3">
        <Avatar className="h-10 w-10">
          <AvatarImage src={item.author.avatar} />
          <AvatarFallback>{item.author.name[0]}</AvatarFallback>
        </Avatar>
        <div className="flex-1">
          <div className="flex items-center gap-2">
            <span className="font-semibold">{item.author.name}</span>
            <span className="text-xs text-muted-foreground">· {item.timeAgo}</span>
          </div>
          <div className="text-xs text-muted-foreground">{item.author.district}</div>
        </div>
        <FollowButton userId={item.author.id} size="sm" />
      </div>
      
      {/* Content */}
      <p className="text-sm leading-relaxed">{item.content}</p>
      
      {/* Images */}
      {item.images?.length > 0 && (
        <div className="grid grid-cols-2 gap-2 rounded-lg overflow-hidden">
          {item.images.slice(0, 4).map((img: any, i: number) => (
            <img 
              key={i} 
              src={img.url} 
              alt={img.alt}
              className="w-full h-48 object-cover"
            />
          ))}
        </div>
      )}
      
      {/* Actions */}
      <div className="flex items-center justify-between pt-3 border-t">
        <button 
          onClick={() => {
            setLiked(!liked)
            setLikes(liked ? likes - 1 : likes + 1)
          }}
          className={`flex items-center gap-1 text-sm ${liked ? 'text-red-500' : 'text-muted-foreground'}`}
        >
          <Heart className={`h-4 w-4 ${liked ? 'fill-red-500' : ''}`} />
          {likes}
        </button>
        
        <button 
          onClick={() => setShowComments(!showComments)}
          className="flex items-center gap-1 text-sm text-muted-foreground hover:text-primary"
        >
          <MessageCircle className="h-4 w-4" />
          {item.commentsCount}
        </button>
        
        <ShareMenu url={`/posts/${item.id}`} title={item.content.slice(0, 50)} />
        
        <BookmarkButton postId={item.id} initialSaved={item.isSaved} />
      </div>
      
      {/* Comments */}
      {showComments && (
        <div className="pt-3 border-t">
          <CommentSection postId={item.id} />
        </div>
      )}
    </article>
  )
}

【8.2 — إدارة النقاط والشراء】

```tsx
// src/components/admin/points-manager.tsx
'use client'
import { useState } from 'react'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Textarea } from '@/components/ui/textarea'
import { Plus, Minus, ShoppingCart, RotateCcw } from 'lucide-react'
import { toast } from 'sonner'

export function PointsManager({ userId }: { userId: string }) {
  const [amount, setAmount] = useState('')
  const [reason, setReason] = useState('')
  
  const handleAdd = async () => {
    await fetch(`/api/admin/users/${userId}/points/add`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ amount: parseInt(amount), reason }),
    })
    toast.success('تم إضافة النقاط')
  }
  
  const handlePurchase = async () => {
    await fetch(`/api/admin/users/${userId}/points/purchase`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ amount: parseInt(amount), pricePaid: 50, method: 'CASH' }),
    })
    toast.success('تم شراء النقاط')
  }
  
  return (
    <div className="glass-card p-6" dir="rtl">
      <h2 className="text-xl font-bold mb-4">إدارة النقاط</h2>
      <div className="space-y-4">
        <Input type="number" placeholder="المبلغ (نقاط)" value={amount} onChange={(e) => setAmount(e.target.value)} dir="ltr" />
        <Textarea placeholder="السبب (إلزامي)" value={reason} onChange={(e) => setReason(e.target.value)} rows={2} />
        <div className="flex gap-2 flex-wrap">
          <Button onClick={handleAdd} className="flex-1"><Plus className="h-4 w-4 ms-2" />إضافة نقاط</Button>
          <Button variant="destructive" className="flex-1"><Minus className="h-4 w-4 ms-2" />خصم نقاط</Button>
          <Button variant="outline" onClick={handlePurchase} className="flex-1"><ShoppingCart className="h-4 w-4 ms-2" />شراء نقاط</Button>
          <Button variant="ghost" className="flex-1"><RotateCcw className="h-4 w-4 ms-2" />استرداد</Button>
        </div>
      </div>
    </div>
  )
}
```
