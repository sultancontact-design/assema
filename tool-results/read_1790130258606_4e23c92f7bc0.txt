   386→function ChangeRoleDialog({
   387→  user,
   388→  open,
   389→  onOpenChange,
   390→}: {
   391→  user: AdminUserRow | null;
   392→  open: boolean;
   393→  onOpenChange: (v: boolean) => void;
   394→}) {
   395→  const [newRole, setNewRole] = React.useState<Role>(user?.role ?? "MEMBER");
   396→
   397→  React.useEffect(() => {
   398→    if (user) setNewRole(user.role);
   399→  }, [user]);
   400→
   401→  if (!user) return null;
   402→
   403→  return (
   404→    <Dialog open={open} onOpenChange={onOpenChange}>
   405→      <DialogContent className="sm:max-w-md">
   406→        <DialogHeader>
   407→          <DialogTitle>تغيير الدور</DialogTitle>
   408→          <DialogDescription>
   409→            تغيير دور {user.fullName} من {ROLE_LABELS[user.role].label}
   410→          </DialogDescription>
   411→        </DialogHeader>
   412→        <div className="space-y-3">
   413→          <Label>الدور الجديد</Label>
   414→          <Select value={newRole} onValueChange={(v) => setNewRole(v as Role)}>
   415→            <SelectTrigger className="h-10 w-full">
   416→              <SelectValue />
   417→            </SelectTrigger>
   418→            <SelectContent>
   419→              {Object.entries(ROLE_LABELS).map(([value, meta]) => (
   420→                <SelectItem key={value} value={value}>
   421→                  {meta.label} — {meta.description}
   422→                </SelectItem>
   423→              ))}
   424→            </SelectContent>
   425→          </Select>
   426→          {newRole !== user.role && (
   427→            <p className="text-xs text-amber-700">
   428→              تنبيه: سيؤثر هذا على صلاحيات المستخدم في النظام.
   429→            </p>
   430→          )}
   431→        </div>
   432→        <DialogFooter>
   433→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   434→            إلغاء
   435→          </Button>
   436→          <Button
   437→            onClick={() => {
   438→              toast.info(
   439→                `تغيير دور ${user.fullName} إلى ${ROLE_LABELS[newRole].label}`
   440→              );
   441→              onOpenChange(false);
   442→            }}
   443→            disabled={newRole === user.role}
   444→          >
   445→            تطبيق
   446→          </Button>
   447→        </DialogFooter>
   448→      </DialogContent>
   449→    </Dialog>
   450→  );
   451→}

   457→export function UsersTable({ users, currentUserRole }: UsersTableProps) {
   458→  // الفلاتر
   459→  const [search, setSearch] = React.useState("");
   460→  const [debouncedSearch, setDebouncedSearch] = React.useState("");
   461→  const [role, setRole] = React.useState<string>("ALL");
   462→  const [status, setStatus] = React.useState<string>("ALL");
   463→
   464→  // الترقيم
   465→  const [page, setPage] = React.useState(1);
   466→
   467→  // النوافذ المنبثقة
   468→  const [viewUser, setViewUser] = React.useState<AdminUserRow | null>(null);
   469→  const [editUser, setEditUser] = React.useState<AdminUserRow | null>(null);
   470→  const [roleUser, setRoleUser] = React.useState<AdminUserRow | null>(null);
   471→  const [deleteUser, setDeleteUser] = React.useState<AdminUserRow | null>(null);
   472→  const [toggleUser, setToggleUser] = React.useState<AdminUserRow | null>(null);
   473→
   474→  // Debounce البحث 300ms
   475→  React.useEffect(() => {
   476→    const t = setTimeout(() => {
   477→      setDebouncedSearch(search.trim().toLowerCase());
   478→      setPage(1);
   479→    }, 300);
   480→    return () => clearTimeout(t);
   481→  }, [search]);
   482→
   483→  // إعادة التصفية عند تغيير الفلاتر
   484→  React.useEffect(() => {
   485→    setPage(1);
   486→  }, [role, status]);
   487→
   488→  // تصفية البيانات في الذاكرة
   489→  const filtered = React.useMemo(() => {
   490→    return users.filter((u) => {
   491→      // البحث
   492→      if (debouncedSearch) {
   493→        const q = debouncedSearch;
   494→        const haystack = `${u.fullName} ${u.email} ${u.phone}`.toLowerCase();
   495→        if (!haystack.includes(q)) return false;
   496→      }
   497→      // الدور
   498→      if (role !== "ALL" && u.role !== role) return false;
   499→      // الحالة
   500→      if (status !== "ALL" && u.status !== status) return false;
   501→      return true;
   502→    });
   503→  }, [users, debouncedSearch, role, status]);
   504→
   505→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   506→  const currentStart = (page - 1) * PAGE_SIZE;
   507→  const currentRows = filtered.slice(currentStart, currentStart + PAGE_SIZE);
   508→
   509→  // هل المستخدم الحالي سوبر أدمن؟ (وحده يغيّر الأدوار ويحذف)
   510→  const canManageRoles = currentUserRole === "SUPER_ADMIN";
   511→
   512→  // إجراء "تعطيل/تفعيل"
   513→  function toggleStatus(u: AdminUserRow) {
   514→    const newStatus = u.status === "ACTIVE" ? "DISABLED" : "ACTIVE";
   515→    toast.info(
   516→      `${newStatus === "ACTIVE" ? "تفعيل" : "تعطيل"} ${u.fullName}`,
   517→      {
   518→        description: `الحالة المستهدفة: ${USER_STATUS_LABELS[newStatus as UserStatus]}`,
   519→      }
   520→    );
   521→    setToggleUser(null);
   522→  }
   523→
   524→  // إجراء "حذف"
   525→  function confirmDelete(u: AdminUserRow) {
   526→    toast.info(`حذف ${u.fullName}`, {
   527→      description: "هذا الإجراء غير معكوس — ",
   528→    });
   529→    setDeleteUser(null);
   530→  }
   531→
   532→  return (
   533→    <div className="space-y-4">
   534→      {/* الفلاتر */}
   535→      <UsersFilters
   536→        search={search}
   537→        onSearch={setSearch}
   538→        role={role}
   539→        onRoleChange={setRole}
   540→        status={status}
   541→        onStatusChange={setStatus}
   542→        resultCount={filtered.length}
   543→      />
   544→
   545→      {/* الجدول */}
   546→      <div className="rounded-lg border border-border bg-card">
   547→        <Table>
   548→          <TableHeader>
   549→            <TableRow className="bg-muted/40 hover:bg-muted/40">
   550→              <TableHead className="text-start text-xs text-muted-foreground">
   551→                المعرّف
   552→              </TableHead>
   553→              <TableHead className="text-start text-xs text-muted-foreground">
   554→                الاسم الكامل
   555→              </TableHead>
   556→              <TableHead className="text-start text-xs text-muted-foreground">
   557→                البريد
   558→              </TableHead>
   559→              <TableHead className="text-start text-xs text-muted-foreground">
   560→                الهاتف
   561→              </TableHead>
   562→              <TableHead className="text-start text-xs text-muted-foreground">
   563→                الحي
   564→              </TableHead>
   565→              <TableHead className="text-start text-xs text-muted-foreground">
   566→                الدور
   567→              </TableHead>
   568→              <TableHead className="text-start text-xs text-muted-foreground">
   569→                الحالة
   570→              </TableHead>
   571→              <TableHead className="text-start text-xs text-muted-foreground">
   572→                التسجيل
   573→              </TableHead>
   574→              <TableHead className="text-start text-xs text-muted-foreground">
   575→                إجراءات
   576→              </TableHead>
   577→            </TableRow>
   578→          </TableHeader>
   579→          <TableBody>
   580→            {currentRows.length === 0 ? (
   581→              <TableRow>
   582→                <TableCell colSpan={9}>
   583→                  <div className="flex flex-col items-center justify-center gap-2 py-12 text-center text-sm text-muted-foreground">
   584→                    <UsersIcon className="size-6 text-muted-foreground/50" strokeWidth={1.5} />
   585→                    <p>لا توجد مستخدمون مطابقون للفلاتر المحدّدة</p>
   586→                  </div>
   587→                </TableCell>
   588→              </TableRow>
   589→            ) : (
   590→              currentRows.map((u) => (
   591→                <TableRow key={u.id} className="text-sm">
   592→                  <TableCell className="font-mono text-[10px] text-muted-foreground">
   593→                    {shortId(u.id)}
   594→                  </TableCell>
   595→                  <TableCell className="font-medium text-foreground">
   596→                    {u.fullName}
   597→                  </TableCell>
   598→                  <TableCell className="text-xs text-muted-foreground" dir="ltr">
   599→                    {u.email}
   600→                  </TableCell>
   601→                  <TableCell className="font-mono text-xs text-muted-foreground" dir="ltr">
   602→                    {u.phone}
   603→                  </TableCell>
   604→                  <TableCell className="text-xs text-foreground">
   605→                    {u.district?.name ?? "—"}
   606→                  </TableCell>
   607→                  <TableCell>
   608→                    <RoleBadge role={u.role} />
   609→                  </TableCell>
   610→                  <TableCell>
   611→                    <StatusBadge status={u.status} />
   612→                  </TableCell>
   613→                  <TableCell className="text-xs text-muted-foreground">
   614→                    {formatDateArabic(u.createdAt)}
   615→                  </TableCell>
   616→                  <TableCell>
   617→                    <DropdownMenu>
   618→                      <DropdownMenuTrigger asChild>
   619→                        <Button
   620→                          variant="ghost"
   621→                          size="icon"
   622→                          className="size-8"
   623→                          aria-label={`إجراءات ${u.fullName}`}
   624→                        >
   625→                          <MoreVertical className="size-4" strokeWidth={1.5} />
   626→                        </Button>
   627→                      </DropdownMenuTrigger>
   628→                      <DropdownMenuContent align="start" className="w-48">
   629→                        <DropdownMenuLabel>إجراءات</DropdownMenuLabel>
   630→                        <DropdownMenuItem onSelect={() => setViewUser(u)}>
   631→                          <Eye className="size-4" strokeWidth={1.5} />
   632→                          <span>عرض</span>
   633→                        </DropdownMenuItem>
   634→                        <DropdownMenuItem onSelect={() => setEditUser(u)}>
   635→                          <Pencil className="size-4" strokeWidth={1.5} />
   636→                          <span>تعديل</span>
   637→                        </DropdownMenuItem>
   638→                        {canManageRoles && (
   639→                          <DropdownMenuItem onSelect={() => setRoleUser(u)}>
   640→                            <ShieldQuestion className="size-4" strokeWidth={1.5} />
   641→                            <span>تغيير الدور</span>
   642→                          </DropdownMenuItem>
   643→                        )}
   644→                        <DropdownMenuSeparator />
   645→                        <DropdownMenuItem onSelect={() => setToggleUser(u)}>
   646→                          <Power className="size-4" strokeWidth={1.5} />
   647→                          <span>
   648→                            {u.status === "ACTIVE" ? "تعطيل" : "تفعيل"}
   649→                          </span>
   650→                        </DropdownMenuItem>
   651→                        {canManageRoles && (
   652→                          <DropdownMenuItem
   653→                            onSelect={() => setDeleteUser(u)}
   654→                            variant="destructive"
   655→                          >
   656→                            <Trash2 className="size-4" strokeWidth={1.5} />
   657→                            <span>حذف</span>
   658→                          </DropdownMenuItem>
   659→                        )}
   660→                      </DropdownMenuContent>
   661→                    </DropdownMenu>
   662→                  </TableCell>
   663→                </TableRow>
   664→              ))
   665→            )}
   666→          </TableBody>
   667→        </Table>
   668→      </div>
   669→
   670→      {/* الترقيم */}
   671→      <div className="flex items-center justify-between">
   672→        <p className="text-xs text-muted-foreground">
   673→          صفحة {page} من {totalPages} · إجمالي {filtered.length}
   674→        </p>
   675→        <div className="flex items-center gap-1">
   676→          <Button
   677→            variant="outline"
   678→            size="sm"
   679→            className="h-9"
   680→            onClick={() => setPage((p) => Math.max(1, p - 1))}
   681→            disabled={page <= 1}
   682→          >
   683→            <ChevronRight className="size-4" strokeWidth={1.5} />
   684→            <span>السابق</span>
   685→          </Button>
   686→          <Button
   687→            variant="outline"
   688→            size="sm"
   689→            className="h-9"
   690→            onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   691→            disabled={page >= totalPages}
   692→          >
   693→            <span>التالي</span>
   694→            <ChevronLeft className="size-4" strokeWidth={1.5} />
   695→          </Button>
   696→        </div>
   697→      </div>
   698→
   699→      {/* النوافذ المنبثقة */}
   700→      <ViewDialog
   701→        user={viewUser}
   702→        open={!!viewUser}
   703→        onOpenChange={(v) => !v && setViewUser(null)}
   704→      />
   705→      <EditDialog
   706→        user={editUser}
   707→        open={!!editUser}
   708→        onOpenChange={(v) => !v && setEditUser(null)}
   709→      />
   710→      <ChangeRoleDialog
   711→        user={roleUser}
   712→        open={!!roleUser}
   713→        onOpenChange={(v) => !v && setRoleUser(null)}
   714→      />
   715→
   716→      {/* تأكيد التعطيل/التفعيل */}
   717→      <AlertDialog
   718→        open={!!toggleUser}
   719→        onOpenChange={(v) => !v && setToggleUser(null)}
   720→      >
   721→        <AlertDialogContent>
   722→          <AlertDialogHeader>
   723→            <AlertDialogTitle>
   724→              {toggleUser?.status === "ACTIVE" ? "تعطيل الحساب" : "تفعيل الحساب"}
   725→            </AlertDialogTitle>
   726→            <AlertDialogDescription>
   727→              {toggleUser?.status === "ACTIVE"
   728→                ? `سيتم تعطيل حساب ${toggleUser?.fullName}. لن يستطيع الدخول حتى يُفعّل مجدداً.`
   729→                : `سيتم تفعيل حساب ${toggleUser?.fullName}.`}
   730→            </AlertDialogDescription>
   731→          </AlertDialogHeader>
   732→          <AlertDialogFooter>
   733→            <AlertDialogCancel>إلغاء</AlertDialogCancel>
   734→            <AlertDialogAction
   735→              onClick={() => toggleUser && toggleStatus(toggleUser)}
   736→            >
   737→              تأكيد
   738→            </AlertDialogAction>
   739→          </AlertDialogFooter>
   740→        </AlertDialogContent>
   741→      </AlertDialog>
   742→
   743→      {/* تأكيد الحذف */}
   744→      <AlertDialog
   745→        open={!!deleteUser}
   746→        onOpenChange={(v) => !v && setDeleteUser(null)}
   747→      >
   748→        <AlertDialogContent>
   749→          <AlertDialogHeader>
   750→            <AlertDialogTitle>تأكيد الحذف</AlertDialogTitle>
   751→            <AlertDialogDescription>
   752→              هل أنت متأكد من حذف حساب {deleteUser?.fullName}؟ هذا الإجراء غير
   753→              معكوس. 
   754→            </AlertDialogDescription>
   755→          </AlertDialogHeader>
   756→          <AlertDialogFooter>
   757→            <AlertDialogCancel>إلغاء</AlertDialogCancel>
   758→            <AlertDialogAction
   759→              className="bg-destructive text-white hover:bg-destructive/90"
   760→              onClick={() => deleteUser && confirmDelete(deleteUser)}
   761→            >
   762→              حذف
   763→            </AlertDialogAction>
   764→          </AlertDialogFooter>
   765→        </AlertDialogContent>
   766→      </AlertDialog>
   767→    </div>
   768→  );
   769→}
