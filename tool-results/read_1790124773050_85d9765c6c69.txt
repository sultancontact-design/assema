   769→function MiniStat({

   801→function ContributionStatusBadge({
   802→  status,
   803→}: {
   804→  status: ContributionStatus;
   805→}) {
   806→  const map: Record<ContributionStatus, string> = {
   807→    PENDING: "bg-amber-100 text-amber-700 border-amber-200",
   808→    CONFIRMED: "bg-emerald-100 text-emerald-700 border-emerald-200",
   809→    REJECTED: "bg-rose-100 text-rose-700 border-rose-200",
   810→    REFUNDED: "bg-slate-100 text-slate-700 border-slate-200",
   811→  };
   812→  return (
   813→    <Badge variant="outline" className={map[status]}>
   814→      {CONTRIBUTION_STATUS_LABELS[status]}
   815→    </Badge>
   816→  );

   821→  children,
